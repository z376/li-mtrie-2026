\section{模型评价}
\subsection{模型优点}

\begin{enumerate}
  \item \textbf{几何建模精确}：将烟幕遮蔽问题严格转化为视线-球-圆柱的求交问题，避免了启发式判定的误差
  \item \textbf{算法效率高}：单次仿真在 1 s 内完成（步长 0.02 s），单次差分进化 100 群体 50 代的总计算时间约 10--30 分钟
  \item \textbf{模型通用性强}：5 个子问题统一在相同的仿真器+优化器框架下，扩展性好
  \item \textbf{物理意义清晰}：所有决策变量都有明确的物理含义（方向、速度、时刻），便于工程实现
  \item \textbf{结果可复现}：使用固定随机种子，结果可复现
\end{enumerate}

\subsection{模型缺点}

\begin{enumerate}
  \item \textbf{优化器局部收敛}：差分进化在大规模问题（25 维）下容易陷入局部最优，可能未达到全局最优
  \item \textbf{目标函数不连续}：视线-球求交的分段特性导致目标函数不光滑，DE 的收敛速度受限
  \item \textbf{未考虑烟幕云团的扩散}：实际烟幕云团会随时间扩散，浓度分布非均匀，本文简化为固定半径的均匀球
  \item \textbf{未考虑风场影响}：实际烟幕云团受风影响会漂移，本文忽略
  \item \textbf{几何模型未考虑部分遮蔽}：当烟幕云团未完全覆盖视线时，可能仍有部分遮蔽效果
  \item \textbf{未考虑导弹末端机动}：实际导弹可能做末端规避，本文假设 M1 始终沿直线飞行
\end{enumerate}

\subsection{模型适用范围}

\begin{itemize}
  \item 适用于烟幕干扰弹投放策略的方案设计
  \item 适用于"几何可见性"主导的遮蔽问题
  \item 对于需要精确数值结果的场景，建议进一步细化仿真步长（0.01 s 以内）
  \item 对超大规模问题（>50 维）建议使用更高效的元启发式算法（如 CMA-ES、PSO）
\end{itemize}
