\subsection*{视线-球相交判定（geometry.py）}

该部分代码使用 AI 工具：Mavis, MiniMax-M3, MiniMax, 2026-08-23

\begin{verbatim}
def segment_sphere(P0, d, C, r):
    """线段 P0 + t*d (t∈[0,1]) 与球 |X-C|=r 的相交区间"""
    d_norm = np.linalg.norm(d)
    if d_norm < 1e-12:
        return None
    u = d / d_norm
    w = P0 - C
    a = 1.0
    b = 2.0 * np.dot(w, u)
    c = np.dot(w, w) - r * r
    disc = b * b - 4 * a * c
    if disc < 0:
        return None
    sq = np.sqrt(disc)
    s0 = (-b - sq) / 2.0
    s1 = (-b + sq) / 2.0
    t_min = max(0.0, s0)
    t_max = min(1.0, s1)
    if t_min >= t_max:
        return None
    return (t_min, t_max)
\end{verbatim}
